const fs=require('fs');const base='docs/references/r4/';const before=JSON.parse(fs.readFileSync(base+'r42-lint-before.json','utf8')),after=JSON.parse(fs.readFileSync(base+'r42-lint-after.json','utf8'));const count=a=>a.reduce((r,x)=>({errors:r.errors+x.errorCount,warnings:r.warnings+x.warningCount}),{errors:0,warnings:0});const sig=x=>x.messages.map(m=>[m.ruleId,m.message,m.severity].join('|')).sort().join('\n');const changes=after.filter(x=>sig(x)!==sig(before.find(b=>b.filePath===x.filePath)||{messages:[]})).map(x=>({file:x.filePath,messages:x.messages}));const r={before:count(before),after:count(after),changes};fs.writeFileSync(base+'r42-lint-comparison.json',JSON.stringify(r,null,2));console.log(JSON.stringify(r,null,2));
