async(page)=>{
const out=[]; const check=(name,ok)=>{out.push({name,ok});if(!ok)throw Error(name);};
await page.getByRole('button',{name:'Novo Cliente',exact:true}).click();
await page.getByRole('button',{name:'Salvar',exact:true}).click();
check('Nome obrigatório não salva',await page.evaluate(()=>!window.r5Calls.some(c=>c.action==='saveCliente')));
const doc=page.getByLabel('CNPJ / CPF',{exact:true});
await doc.fill('123'); check('Documento incompleto',await page.getByText(/Documento incompleto/).isVisible());
await doc.fill('11111111111');check('CPF repetido inválido',await page.getByText('CPF inválido (verifique os dígitos)').isVisible());
await doc.fill('52998224725');check('CPF válido e máscara',await doc.inputValue()==='529.982.247-25'&&await page.getByText('CPF inválido (verifique os dígitos)').count()===0);
await doc.fill('12345678000190');check('CNPJ inválido',await page.getByText('CNPJ inválido (dígitos verificadores incorretos)').isVisible());
await doc.fill('12345678000195');check('CNPJ válido e máscara',await doc.inputValue()==='12.345.678/0001-95'&&await page.getByRole('button',{name:'Buscar dados',exact:true}).isVisible());
await page.getByRole('button',{name:'Buscar dados',exact:true}).click();
await page.getByLabel('Nome / Empresa *',{exact:true}).waitFor();
check('Consulta simulada preenche',await page.getByLabel('Nome / Empresa *',{exact:true}).inputValue()==='EXEMPLO DIGITAL');
check('Telefone da consulta formatado',await page.getByLabel('Telefone',{exact:true}).inputValue()==='(48) 3333-4444');
check('Campos empresariais exibidos',await page.getByLabel('Razão Social',{exact:true}).inputValue()==='EMPRESA EXEMPLO SINTETICA LTDA');
await page.getByLabel('Nome / Empresa *',{exact:true}).fill('Nome preservado');
await page.getByRole('button',{name:'Buscar dados',exact:true}).click();
const conflict=page.getByRole('region',{name:'Confirmar dados do CNPJ'});
await conflict.waitFor();
check('Conflito não sobrescreve',await page.getByLabel('Nome / Empresa *',{exact:true}).inputValue()==='Nome preservado');
await conflict.getByRole('button',{name:'Cancelar',exact:true}).click();
check('Cancelar conflito mantém dados',await conflict.count()===0&&await page.getByLabel('Nome / Empresa *',{exact:true}).inputValue()==='Nome preservado');
await page.getByRole('button',{name:'Buscar dados',exact:true}).click();
await conflict.getByRole('button',{name:'Aplicar',exact:true}).click();
check('Aplicar conflito atualiza',await page.getByLabel('Nome / Empresa *',{exact:true}).inputValue()==='EXEMPLO DIGITAL');
await page.getByRole('button',{name:'Salvar',exact:true}).click();
check('Criação somente em memória',await page.evaluate(()=>window.r5Store.getState().data.clientes.length===13&&window.r5Calls.filter(c=>c.action==='saveCliente').length===1));
await page.getByRole('button',{name:'TechVibe Soluções Digitais',exact:true}).click();
await page.getByRole('button',{name:'Editar cliente',exact:true}).click();
await page.getByLabel('Telefone',{exact:true}).fill('(48) 99999-0000');
await page.getByRole('button',{name:'Salvar',exact:true}).click();
check('Edição pelo editingId sem duplicar',await page.evaluate(()=>{const d=window.r5Store.getState().data.clientes;return d.length===13&&d.find(x=>x.id==='cli-001').telefone==='(48) 99999-0000';}));
await page.getByRole('button',{name:'TechVibe Soluções Digitais',exact:true}).click();
await page.getByRole('button',{name:'Editar cliente',exact:true}).click();
await page.getByLabel('Nome / Empresa *',{exact:true}).fill('Não salvar');
await page.getByRole('button',{name:'Cancelar',exact:true}).click();
check('Cancelar edição descarta rascunho',await page.evaluate(()=>window.r5Store.getState().data.clientes.find(x=>x.id==='cli-001').nome==='TechVibe Soluções Digitais'));
await page.evaluate(v=>window.r51Results.push(...v),out);
return out;
}
